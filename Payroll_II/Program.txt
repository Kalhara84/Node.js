// Line 9: Core request function
export async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
    const headers: Record<string, string> = { 'Content-Type': 'application/json' };

    // Attach Bearer token from localStorage
    const token = localStorage.getItem('payroll_token');
    if (token) headers.Authorization = `Bearer ${token}`;

    // Send HTTP Request to Backend API
    const res = await fetch('http://localhost:3001/api' + path, {
        ...init,
        headers
    });

    const data = await res.json();
    if (!res.ok) throw new Error(data.message);
    return data;
}
```

const router = Router();

// GET all programs -> checks 'view' permission -> calls controller.programs
router.get('/programs', requireScreen('programs'), wrap(AdministrationController.programs));

// POST create program -> checks 'add' permission -> calls controller.createProgram
router.post('/programs', requireScreen('programs', 'add'), wrap(AdministrationController.createProgram));

// PUT update program -> checks 'amend' permission -> calls controller.updateProgram
router.put('/programs/:id', requireScreen('programs', 'amend'), wrap(AdministrationController.updateProgram));

// DELETE program -> checks 'delete' permission -> calls controller.deleteProgram
router.delete('/programs/:id', requireScreen('programs', 'delete'), wrap(AdministrationController.deleteProgram));

// Error handler for validation and duplicate constraint errors

export default router;

// Line 151: Validates program inputs and returns formatted values
function programValues(body: Record<string, unknown>) {
    const module = integer(body.prgmodsno, 'Module');
    const submodule = integer(body.prgsmdsno, 'Submodule');

    if (!['S', 'M'].includes(String(body.prglvl))) {
        fail('Choose Single or Multi level');
    }

    const name = string(body.prgnam, 'Name', 35, true);
    if (!/^[a-z][a-z0-9_-]*$/.test(name) || ['admin', 'constructor', 'prototype'].includes(name)) {
        fail('Use a program key starting with a lowercase letter, followed by lowercase letters, numbers, hyphens or underscores');
    }

    return {
        module,
        submodule,
        values: [
            module,
            submodule,
            string(body.prgorder, 'Order', 5, true),
            body.prglvl,
            name,
            string(body.prgdes, 'Description', 60, true)
        ]
    };
}

// Line 182: SQL VALIDATION - Checks if Module (32) and Submodule (33) exist in database
async function checkProgramScope(module: number, submodule: number): Promise<void> {
    const result = await pool.query(
        'SELECT syssno FROM a_sysmast WHERE (syssno=$1 AND sysrtp=32) OR (syssno=$2 AND sysrtp=33)',
        [module, submodule]
    );
    if (result.rowCount !== 2) {
        fail('Choose a valid module and submodule');
    }
}
```

// Line 253: AdministrationController.programs
async programs(_req: Request, res: Response): Promise<void> {
    const result = await pool.query(`
        SELECT p.*, m.sysdes AS module, s.sysdes AS submodule
        FROM prgmast p
        LEFT JOIN a_sysmast m ON m.syssno = p.prgmodsno
        LEFT JOIN a_sysmast s ON s.syssno = p.prgsmdsno
        WHERE p.prgapp='payroll-web'
        ORDER BY p.prgmodsno, p.prgsmdsno, p.prgorder, p.prgsno
    `);
    res.json(result.rows);
}
```

async createProgram(req: Request, res: Response): Promise<void> {
    const { module, submodule, values } = programValues(req.body ?? {});

    const result = await pool.query(`
        INSERT INTO prgmast(
            prgmodsno, prgsmdsno, prgorder, prglvl, prgnam, prgdes,

        )
        VALUES($1, $2, $3, $4, $5, $6, $7, CURRENT_DATE, LOCALTIME, 'payroll-web')
        RETURNING *

    res.status(201).json(result.rows[0]); // Returns 201 Created with saved record
}

async updateProgram(req: Request, res: Response): Promise<void> {
    const id = integer(req.params.id, 'Program');
    const { module, submodule, values } = programValues(req.body ?? {});
    await checkProgramScope(module, submodule); // SQL Validation

    const result = await pool.query(`
        UPDATE prgmast
        SET prgmodsno=$1, prgsmdsno=$2, prgorder=$3, prglvl=$4, prgnam=$5, prgdes=$6,
            prgucdame=$7, prgameddt=CURRENT_DATE, prgametime=LOCALTIME
        WHERE prgsno=$8 AND prgapp='payroll-web'
        RETURNING *
    `, [...values, req.user!.usr, id]);

    const program = result.rows[0] ?? null;
    res.status(program ? 200 : 404).json(program ?? { message: 'Program not found' });
}

D. DELETE PROGRAM (TRANSACTION):

// Line 299: AdministrationController.deleteProgram
async deleteProgram(req: Request, res: Response): Promise<void> {
    const id = integer(req.params.id, 'Program');
    const client = await pool.connect();
    try {
        await client.query('BEGIN');
        await client.query('LOCK TABLE prgmast IN SHARE ROW EXCLUSIVE MODE');

        const program = await client.query(
            "SELECT prgsno FROM prgmast WHERE prgsno=$1 AND prgapp='payroll-web'", 
            [id]
        );
        if (program.rowCount) {
            // 1. Delete associated user rights first
            await client.query('DELETE FROM usrmast WHERE usrprgsno=$1', [id]);
            // 2. Delete the program
            await client.query("DELETE FROM prgmast WHERE prgsno=$1 AND prgapp='payroll-web'", [id]);
        }
        await client.query('COMMIT');
        res.status(program.rowCount ? 200 : 404).json({ message: program.rowCount ? 'Program deleted' : 'Program not found' });
    } catch (error) {
        await client.query('ROLLBACK');
        throw error;
    } finally {
        client.release();
    }
}
```

// Line 196: AdministrationController.handleError
handleError(err: unknown, _req: Request, res: Response, next: NextFunction): void {
    const e = err as { status?: number; code?: string; message?: string };
    if (e.status === 400) {
        res.status(400).json({ message: e.message });
        return;
    }
    // PostgreSQL error code 23505 = Unique key violation
    if (e.code === '23505') {
        res.status(409).json({ message: 'This code or module/submodule/order already exists' });
        return;
    }
    next(err);
}
